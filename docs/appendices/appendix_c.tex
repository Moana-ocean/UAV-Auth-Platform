\section{Data Dictionary}
\label{app:data-dictionary}

Table~\ref{tab:app-obs-fields} defines the principal per-request fields in \texttt{observations.csv}. Full definitions, latency boundaries and summary-statistic methods are documented in \texttt{docs/METRICS.md} (copied to \texttt{export/00\_METRICS.md}).

\begin{table}[htbp]
    \centering
    \caption{Selected per-request observation fields}
    \label{tab:app-obs-fields}
    \small
    \begin{tabular}{|p{0.28\textwidth}|p{0.62\textwidth}|}
        \hline
        \textbf{Field} & \textbf{Meaning} \\
        \hline
        \texttt{run\_id} & Unique experimental run identifier \\
        \hline
        \texttt{mechanism} & \texttt{x509} or \texttt{blockchain} \\
        \hline
        \texttt{scenario} & Stable scenario id (e.g.\ \texttt{valid\_active}, \texttt{replay}) \\
        \hline
        \texttt{repetition} & 0-based measured repetition index \\
        \hline
        \texttt{expected\_outcome} / \texttt{observed\_outcome} & Reason codes; match required for correctness \\
        \hline
        \texttt{expectation\_met} & \texttt{true} iff expected and observed codes match \\
        \hline
        \texttt{decision\_latency\_ms} & GCS receipt of complete request $\rightarrow$ accept/reject decision \\
        \hline
        \texttt{identity\_lookup\_ns} & Certificate/CRL load or registry retrieval only \\
        \hline
        \texttt{contract\_call\_ns} & Read-only \texttt{eth\_call} duration (blockchain only) \\
        \hline
        \texttt{audit\_confirmation\_ns} & Audit tx submit $\rightarrow$ confirmation (blockchain, audit on) \\
        \hline
        \texttt{warmup} & \texttt{true} if excluded from default summaries \\
        \hline
        \texttt{tx\_hash}, \texttt{gas\_used} & Present for audit or setup transactions \\
        \hline
    \end{tabular}
\end{table}

\begin{table}[htbp]
    \centering
    \caption{Selected run-level summary fields}
    \label{tab:app-summary-fields}
    \small
    \begin{tabular}{|p{0.28\textwidth}|p{0.62\textwidth}|}
        \hline
        \textbf{Field} & \textbf{Meaning} \\
        \hline
        \texttt{n\_warmup\_excluded} & Warm-up rows omitted from default analysis \\
        \hline
        \texttt{false\_acceptances} & Adversarial request incorrectly accepted \\
        \hline
        \texttt{false\_rejections} & Valid scenario incorrectly rejected \\
        \hline
        \texttt{attack\_rejection\_rate\_pct} & Malicious requests rejected / malicious submitted $\times 100$ \\
        \hline
        \texttt{failure\_rate\_pct} & Timeouts and internal errors only; expected security rejections excluded \\
        \hline
        \texttt{mean\_ms}, \texttt{p95\_ms}, \texttt{ci95\_low\_ms} & Decision-latency statistics (warm-up excluded) \\
        \hline
        \texttt{latency\_set} & \texttt{all\_finished} or \texttt{accepted\_only} filter \\
        \hline
    \end{tabular}
\end{table}
